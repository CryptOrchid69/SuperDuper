| Keyword | Purpose | Example |
| --- | --- | --- |
| False | Boolean value (not true) | is_active = False |
| None | Empty/null value | result = None |
| True | Boolean value (is true) | is_active = True |
| and | Logical AND | if x > 5 and y < 10: |
| as | Give an alias/nickname | import numpy as np |
| assert | Test if something is true | assert x > 0 |
| async | Define async function | async def fetch(): |
| await | Wait for async result | result = await fetch() |
| break | Exit a loop | if x == 5: break |
| class | Define a class | class Dog: |
| continue | Skip to next loop | if x == 5: continue |
| def | Define a function | def greet(): |
| del | Delete a variable | del my_var |
| elif | Else if condition | elif x > 5: |
| else | Default condition | else: |
| except | Handle errors | except ValueError: |
| finally | Always execute | finally: |
| for | Loop through items | for i in range(10): |
| from | Import specific item | from math import sqrt |
| global | Declare global variable | global x |
| if | Conditional statement | if x > 5: |
| import | Import a module | import math |
| in | Check membership | if "a" in "apple": |
| is | Check identity | if x is y: |
| lambda | Anonymous function | square = lambda x: x**2 |
| nonlocal | Declare non-local variable | nonlocal x |
| not | Logical NOT | if not x: |
| or | Logical OR | if x > 5 or y < 10: |
| pass | Do nothing (placeholder) | if x: pass |
| raise | Raise an error | raise ValueError() |
| return | Return from function | return result |
| try | Try/catch errors | try: |
| while | Loop while true | while x < 10: |
| with | Context manager | with open(file) as f: |
| yield | Generator function | yield x |
